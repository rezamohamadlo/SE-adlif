"""One-off CPU primary-validation diagnosis; no training, CUDA or official tests."""
import contextlib,hashlib,io,json,time
from pathlib import Path
import h5py,numpy as np,torch
from hydra.utils import instantiate
from omegaconf import OmegaConf
from models.pl_module import build_training_model
from module.study_control import forward_identity,canonical
from project_paths import ROOT,local_path

torch.set_num_threads(2);torch.set_num_interop_threads(1)
torch.set_float32_matmul_precision('high');torch.manual_seed(42)
OUT=ROOT/'results/experiments/complementary_speaker_dev_v2'
SOURCE=OUT/'B002_PRIMARY_CPU_DIAGNOSTIC_SOURCE.txt'
SOURCE_SHA=hashlib.sha256(SOURCE.read_bytes()).hexdigest()
TARGET=OUT/'B002_PRIMARY_CPU_DIAGNOSTICS.json'
assert not TARGET.exists(),'Preserve existing diagnostic summary'
manifest=json.loads((OUT/'primary_0_3/SPLIT_MANIFEST.json').read_text())
train_file=local_path('data/SHDWrapper/shd_train.h5',must_exist=True)
with train_file.open('rb') as stream:assert hashlib.file_digest(stream,'sha256').hexdigest()==manifest['dataset_identity']['train_sha256']
with h5py.File(train_file,'r') as f:
    raw_labels=f['labels'][:];raw_speakers=f['extra/speaker'][:]
records=[];rows_by_model={}
for name in ['SE_adLIF','LS_SE_adLIF']:
    run=OUT/'b002_label_smoothing_v1/primary_0_3'/name/'SHD/seed_42'
    receipt=json.loads((run/'stages/030/RESULT.json').read_text())
    checkpoint=local_path(receipt['best_checkpoint'],must_exist=True)
    assert hashlib.sha256(checkpoint.read_bytes()).hexdigest()==receipt['best_checkpoint_sha256']
    assert json.loads((run/'MODEL_SOURCE_IDENTITY.json').read_text())==forward_identity('SHD')
    state=torch.load(checkpoint,map_location='cpu',weights_only=False)
    assert state['epoch']==receipt['selected_epoch_index']
    original=OmegaConf.to_container(OmegaConf.create(state['hyper_parameters']['cfg']),resolve=True)
    assert canonical(original)==canonical(json.loads((run/'FULL_RUN_CONFIG.json').read_text()))
    assert original['dataset']['validation_only_data'] is True and original['run_test'] is False
    assert original['dataset']['speaker_split_fingerprint']==manifest['fingerprint']
    cfg=OmegaConf.create(original);cfg.device='cpu';cfg.dataset.num_workers=0
    dm=instantiate(cfg.dataset,data_path=str(local_path(cfg.dataset.data_path)));assert not hasattr(dm,'data_test')
    assert list(dm.data_val.indices)==manifest['subsets']['validation']['indices']
    with contextlib.redirect_stdout(io.StringIO()):model=build_training_model(cfg)
    model.load_state_dict(state['state_dict'],strict=True);model.cpu().eval()
    assert not torch.cuda.is_initialized(),'CPU diagnostic must not initialize CUDA'
    probabilities=[];labels=[];losses=[];start=time.perf_counter()
    with torch.inference_mode():
        for x,y,blocks in dm.val_dataloader():
            assert x.device.type==y.device.type==blocks.device.type=='cpu'
            reduced,loss,_=model.process_predictions_and_compute_losses(model(x),y,blocks)
            mask=y.flatten().ne(-1);assert int(mask.sum())==len(x)
            scores=reduced[mask].float();target=y.flatten()[mask].long()
            probabilities.append(torch.softmax(scores,-1).numpy());labels.append(target.numpy())
            losses.append(torch.nn.functional.cross_entropy(scores,target,reduction='none').numpy())
    elapsed=time.perf_counter()-start
    p=np.concatenate(probabilities);y=np.concatenate(labels);ce=np.concatenate(losses)
    indices=np.asarray(dm.data_val.indices,dtype=np.int64);speaker=raw_speakers[indices]
    assert len(y)==len(indices)==1635 and np.array_equal(y,raw_labels[indices])
    assert set(np.unique(speaker))=={0,3} and np.isfinite(p).all() and np.isfinite(ce).all()
    pred=p.argmax(1);correct=pred==y;confidence=p.max(1)
    bin_ids=np.minimum((confidence*15).astype(int),14);bins=[];ece=0.
    for i in range(15):
        mask=bin_ids==i;n=int(mask.sum())
        acc=float(correct[mask].mean()) if n else None;conf=float(confidence[mask].mean()) if n else None
        if n:ece+=n/len(y)*abs(acc-conf)
        bins.append(dict(bin=i,count=n,accuracy=acc,mean_confidence=conf))
    confusion=np.zeros((20,20),dtype=np.int64);np.add.at(confusion,(y,pred),1)
    counts=confusion.sum(1);recall=np.divide(np.diag(confusion),counts,out=np.zeros(20,dtype=float),where=counts>0)
    onehot=np.eye(20)[y]
    record=dict(model=name,role='primary_development_CPU_diagnostic',samples=len(y),
        checkpoint=receipt['best_checkpoint'],checkpoint_sha256=receipt['best_checkpoint_sha256'],
        selected_completed_epoch=receipt['selected_epoch_index']+1,
        authoritative_GPU_selected_accuracy=receipt['best_validation_accuracy'],
        CPU_recomputed_accuracy=float(correct.mean()),CPU_minus_GPU_accuracy_pp=100*(float(correct.mean())-receipt['best_validation_accuracy']),
        CPU_cross_entropy=float(ce.mean()),ece15=float(ece),brier_sum_over_classes=float(np.square(p-onehot).sum(1).mean()),
        mean_confidence=float(confidence.mean()),mean_wrong_confidence=float(confidence[~correct].mean()) if (~correct).any() else None,
        macro_class_recall=float(recall[counts>0].mean()),class_counts=counts.tolist(),class_recall=recall.tolist(),confusion_matrix=confusion.tolist(),
        per_speaker={str(int(s)):dict(recordings=int((speaker==s).sum()),accuracy=float(correct[speaker==s].mean()),cross_entropy=float(ce[speaker==s].mean()),classes_present=np.unique(y[speaker==s]).tolist()) for s in np.unique(speaker)},
        calibration_bins=bins,confidence_definition='CE-implied softmax of unchanged temporally aggregated class scores; no probability fitting',
        runtime=dict(device='cpu',threads=2,interop_threads=1,validation_batch=int(cfg.dataset.validation_batch_size),workers=0,seconds=elapsed,CUDA_initialized=torch.cuda.is_initialized()),
        source_code=SOURCE.relative_to(ROOT).as_posix(),source_sha256=SOURCE_SHA,model_source_identity=forward_identity('SHD'),
        official_test_opened=False,selection_scores_replaced=False,new_hyperparameters_selected=False)
    assert int(confusion.sum())==len(y) and sum(b['count'] for b in bins)==len(y)
    directory=run/'diagnostic_rows';directory.mkdir(exist_ok=True)
    raw_file=directory/'stage030_primary_CPU_predictions.npz';assert not raw_file.exists()
    with raw_file.open('xb') as stream:np.savez_compressed(stream,row_indices=indices,speaker_ids=speaker,targets=y,predictions=pred,probabilities=p,cross_entropy=ce)
    record['prediction_rows']=raw_file.relative_to(ROOT).as_posix();record['prediction_rows_sha256']=hashlib.sha256(raw_file.read_bytes()).hexdigest()
    record_file=run/'VALIDATION_CPU_DIAGNOSTICS_STAGE030.json';assert not record_file.exists()
    record_file.write_text(json.dumps(record,indent=2)+'\n')
    records.append(record);rows_by_model[name]=(indices,speaker,y,pred,correct)
    print('[CPU_DIAGNOSTIC] '+json.dumps({k:record[k] for k in ['model','CPU_recomputed_accuracy','CPU_minus_GPU_accuracy_pp','CPU_cross_entropy','ece15','brier_sum_over_classes','mean_wrong_confidence','per_speaker','runtime']}),flush=True)
    del model,dm,state,x,y,blocks,reduced,scores
base=rows_by_model['SE_adLIF'];candidate=rows_by_model['LS_SE_adLIF']
assert all(np.array_equal(base[i],candidate[i]) for i in (0,1,2))
paired={}
for label,mask in [('all',np.ones(len(base[0]),dtype=bool)),('speaker0',base[1]==0),('speaker3',base[1]==3)]:
    b=base[4][mask];c=candidate[4][mask]
    paired[label]=dict(recordings=len(b),both_correct=int((b&c).sum()),baseline_only_correct=int((b&~c).sum()),LS_only_correct=int((~b&c).sum()),both_wrong=int((~b&~c).sum()),LS_minus_Baseline_pp=100*float(c.mean()-b.mean()))
result=dict(status='completed_primary_CPU_validation_diagnostic',records=records,paired_correctness=paired,
    interpretation='Primary historically tuned0/3development only, one training seed and validation-selected checkpoints. CPU backend differs from original GPU selection; report differences, never overwrite selection scores or count as an independent result. Calibration and class/speaker diagnostics do not establish causality, synergy, official-test gain or a new tuning target.',
    next_action='Keep SSC/ECG transfer gates; no epsilon/pair tuning or new heavy combination based on this diagnostic alone.',official_test_opened=False)
TARGET.write_text(json.dumps(result,indent=2)+'\n');print('[CPU_DIAGNOSTIC_COMPLETE] '+json.dumps(paired),flush=True)

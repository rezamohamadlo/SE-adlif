"""One-off real-data disposable SSC confirmation; no main Trainer, queue or checkpoints."""
from pathlib import Path
from datetime import datetime
import contextlib, gc, hashlib, io, json, math, statistics, time
import hydra
from hydra import compose, initialize_config_dir
from omegaconf import OmegaConf
import pytorch_lightning as pl
import torch
from models.pl_module import build_training_model
from module.study_control import forward_identity
from project_paths import ROOT, local_path, portable_config

OUT=ROOT/'results/experiments/complementary_speaker_dev_v2'
TARGET=OUT/'B002_SSC_BATCH1024_CONFIRMATION.json'
assert not TARGET.exists(), 'Preserve existing calibration evidence'
assert torch.cuda.is_available()
torch.set_float32_matmul_precision('high')
device=torch.device('cuda:0')
cal=json.loads((OUT/'SSC_TRAINING_CALIBRATION.json').read_text())
assert cal['selected_batch']==1024 and len(cal['source_indices'])==1024
records=[]; data=None;cpu_train=None;cpu_val=None
for experiment in ['SSC_SE_adLIF','SSC_LS_SE_adLIF']:
    pl.seed_everything(117,workers=True)
    with initialize_config_dir(config_dir=str(ROOT/'config'),version_base=None):
        cfg=compose(config_name='main',overrides=['experiment='+experiment,'+comparison=SSC','random_seed=117','batch_size=1024','dataset.batch_size=1024','dataset.num_workers=0','++dataset.spawn_safe_transforms=true','++dataset.validation_only_data=true','++dataset.validation_batch_size=1024','run_test=false','early_stopping=false'])
        cfg=OmegaConf.create(OmegaConf.to_container(cfg,resolve=True))
    assert cfg.lr==.006 and cfg.n_epochs==40 and cfg.dropout==.15 and cfg.loss_agg=='mean_membrane_potentials'
    if data is None:
        cfg.dataset.data_path=str(local_path(cfg.dataset.data_path))
        data=hydra.utils.instantiate(cfg.dataset);data.prepare_data();data.setup('fit')
        assert not hasattr(data,'data_test') and len(data.data_train)==75466 and len(data.data_val)==9981
        cpu_train=[data.data_train[i] for i in cal['source_indices']]
        cpu_val=[data.data_val[i] for i in range(1024)]
    with contextlib.redirect_stdout(io.StringIO()):model=build_training_model(cfg).to(device).train()
    configured=model.configure_optimizers();optimizer=configured['optimizer'] if isinstance(configured,dict) else configured
    batch=tuple(t.to(device) for t in data.collate_fn(cpu_train))
    assert batch[0].shape[0]==1024 and batch[0].shape[1]>=300
    times=[];losses=[]
    for step in range(5):
        optimizer.zero_grad(set_to_none=True);torch.cuda.synchronize()
        if step==2:torch.cuda.reset_peak_memory_stats()
        start=time.perf_counter();outputs=model(batch[0]);_,loss,_=model.process_predictions_and_compute_losses(outputs,batch[1],batch[2]);loss.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(),float(cfg.gradient_clip_val));optimizer.step();model.on_train_batch_end(None,batch,step)
        torch.cuda.synchronize();elapsed=time.perf_counter()-start;losses.append(float(loss.detach().cpu()))
        if step>=2:times.append(elapsed)
        assert math.isfinite(losses[-1])
        assert all(torch.isfinite(p.grad).all().item() for p in model.parameters() if p.grad is not None)
    reserved=torch.cuda.max_memory_reserved();allocated=torch.cuda.max_memory_allocated();headroom=1-reserved/torch.cuda.get_device_properties(device).total_memory
    assert headroom>=.15
    train={'warmup_steps':2,'measured_steps':3,'step_seconds':statistics.mean(times),'samples_per_second':1024/statistics.mean(times),'peak_allocated_gib':allocated/2**30,'peak_reserved_gib':reserved/2**30,'headroom_fraction':headroom,'timesteps':int(batch[0].shape[1]),'updates_per_epoch':75466//1024,'estimated_epoch_train_seconds':statistics.mean(times)*(75466//1024),'finite_losses_and_gradients':True,'losses':losses}
    del optimizer,configured,batch,outputs,loss;gc.collect();torch.cuda.empty_cache();model.eval()
    probes=[];predictions={}
    for size in [256,1024]:
        def validation_pass():
            count=0;weighted=0.;pred=[];torch.cuda.synchronize();start=time.perf_counter()
            with torch.inference_mode():
                for offset in range(0,1024,size):
                    x,y,b=(t.to(device) for t in data.collate_fn(cpu_val[offset:offset+size]))
                    reduced,ce,_=model.process_predictions_and_compute_losses(model(x),y,b)
                    mask=y.flatten().ne(-1);n=int(mask.sum());count+=n;weighted+=float(ce)*n;pred.extend(reduced[mask].argmax(-1).cpu().tolist())
            torch.cuda.synchronize();assert count==1024
            return time.perf_counter()-start,weighted/count,pred
        validation_pass();torch.cuda.reset_peak_memory_stats();passes=[validation_pass(),validation_pass()]
        assert passes[0][2]==passes[1][2]
        predictions[size]=passes[1][2]
        probes.append({'batch':size,'warmup_passes':1,'measured_passes':2,'validation_probe_records':1024,'seconds':statistics.mean(v[0] for v in passes),'loss':passes[1][1],'peak_reserved_gib':torch.cuda.max_memory_reserved()/2**30})
    assert predictions[256]==predictions[1024]
    assert abs(probes[0]['loss']-probes[1]['loss'])<1e-5
    record={'experiment':experiment,'training_objective':cfg.get('training_objective','classification'),'epsilon':cfg.get('label_smoothing_epsilon',0),'train':train,'validation_probes':probes,'validation_changed_predictions':0,'validation_loss_difference':abs(probes[0]['loss']-probes[1]['loss']),'config':portable_config(OmegaConf.to_container(cfg,resolve=True))}
    records.append(record);print('[B002_SSC_CONFIRMATION] '+json.dumps(record),flush=True)
    del model;gc.collect();torch.cuda.empty_cache()
result={'status':'verified','created_at':datetime.now().astimezone().isoformat(),'models':records,'training_batch':1024,'validation_batch':1024,'lr':.006,'lr_scaled':False,'num_workers_main':8,'worker_evidence':'SSC_WORKER_CALIBRATION.json','training_source_indices':'SSC_TRAINING_CALIBRATION.json/source_indices','validation_probe_indices':'official valid rows0..1023, unchanged order; bounded numerical probe, not accuracy evidence','existing_full_validation_calibration':'SSC_VALIDATION_CALIBRATION.json','source_identity':forward_identity('SSC'),'disposable_seed':117,'disposable_states_discarded':True,'main_weights_or_optimizer_reused':False,'official_test_opened':False,'note':'Confirms selected batches under current objective. Existing full batch/worker grids are preserved and not relabeled.'}
TARGET.write_text(json.dumps(result,indent=2)+'\n')
print('[CALIBRATION_COMPLETE] '+str(TARGET.relative_to(ROOT)),flush=True)

def differences(a,b,prefix=''):
    if isinstance(a,dict) and isinstance(b,dict):
        result={}
        for key in set(a)|set(b):result.update(differences(a.get(key),b.get(key),prefix+'.'+key if prefix else key))
        return result
    return {} if a==b else {prefix:dict(RDrop=a,control=b)}
diff=differences(canonical(configs['RDROP_SE_adLIF']),canonical(configs['TWO_PASS_SE_adLIF']))
allowed={'model_variant','exp_name','logdir','paper_baseline_deviations','study_arm','study_hypothesis','dropout_consistency_alpha'}
assert set(diff).issubset(allowed),set(diff)-allowed

                if key not in item['tensors'] or any(item['tensors'][key][field] != value[field]

    run = RUNS/name/'SHD/seed_42'

    full = json.loads((run/'FULL_RUN_CONFIG.json').read_text())

    assert all(math.isfinite(float(r[k])) for r in val for k in ('val_acc', 'val_loss'))

        assert state['global_step'] == 6*(state['epoch']+1)

    cfg = OmegaConf.create(full)

    pl.seed_everything(42, workers=True)

    data = hydra.utils.instantiate(cfg.dataset)

    predicted = []
    target_rows = []
    probability_rows = []
    weighted_loss = 0.
    count = 0
    with torch.inference_mode():

            x, y, blocks = (t.cuda() for t in cpu)
            reduced, loss, _ = model.process_predictions_and_compute_losses(model(x), y, blocks)
            mask = y.flatten().ne(-1)

            assert n == len(x)
            scores = reduced[mask].float()

            target_rows.extend(y.flatten()[mask].cpu().tolist())

            assert abs(value('train_loss_epoch')-value('train_supervised_loss')-value('train_dropout_symmetric_kl')) < 1e-4
    record = dict(model=name, selected_validation_accuracy=float(selected['val_acc']),

def differences(a,b,path=''):
    if isinstance(a,dict) and isinstance(b,dict):
        result = {}
        for key in set(a)|set(b):
            result.update(differences(a.get(key),b.get(key),path+'.'+key if path else key))
        return result
    return {} if a==b else {path:dict(baseline=a,candidate=b)}

"""CPU-only technical integration, synthetic inputs; no datasets or training."""
import os
os.environ['CUDA_VISIBLE_DEVICES'] = ''
import contextlib, hashlib, importlib.util, io, json, time
from pathlib import Path
from datetime import datetime, timezone
import torch
from omegaconf import OmegaConf
from models.pl_module import build_training_model
from project_paths import ROOT, local_path

OUT = ROOT/'results/experiments/complementary_speaker_dev_v2/cost_draft/rdrop_capture_integration_v1'
SOURCE = OUT/'SOURCE.txt'
TARGET = OUT/'VERIFICATION.json'
assert not TARGET.exists(), 'Preserve existing technical evidence'
registered = json.loads((OUT/'PREREGISTRATION.json').read_text())
assert hashlib.sha256(SOURCE.read_bytes()).hexdigest() == registered['source_sha256']
capture_path = ROOT/registered['capture_source']
assert hashlib.sha256(capture_path.read_bytes()).hexdigest() == registered['capture_sha256']
spec = importlib.util.spec_from_file_location('unapplied_capture_draft', capture_path)
capture = importlib.util.module_from_spec(spec)
spec.loader.exec_module(capture)
torch.set_num_threads(2)
assert not torch.cuda.is_initialized()
started = time.perf_counter()
records = []
for name in registered['models']:
    run = ROOT/'results/experiments/complementary_speaker_dev_v2/c005_dropout_consistency_v1/primary_0_3'/name/'SHD/seed_42'
    receipt = json.loads((run/'stages/030/RESULT.json').read_text())
    path = local_path(receipt['best_checkpoint'], must_exist=True)
    assert hashlib.sha256(path.read_bytes()).hexdigest() == receipt['best_checkpoint_sha256']
    state = torch.load(path, map_location='cpu', weights_only=False)
    cfg = OmegaConf.create(state['hyper_parameters']['cfg'])
    with contextlib.redirect_stdout(io.StringIO()):
        model = build_training_model(cfg).cpu().eval()
    model.load_state_dict(state['state_dict'], strict=True)
    assert all(t.device.type == 'cpu' for t in model.parameters())
    # Trainer logging is outside this technical scope. Keep the real metric
    # update method and real test_step/aggregation/loss; silence only log sinks.
    model.log = lambda *a, **k: None
    model.log_dict = lambda *a, **k: None
    if hasattr(model, 'training_terms'):
        def forbid_training(*a, **k):
            raise AssertionError('Training-only objective must not execute')
        model.training_terms = forbid_training
    width = int(cfg.l1.input_size)
    x = ((torch.arange(4*12*width).reshape(4,12,width) % 13) == 0).float()
    x[:,-2:] = 0
    y = torch.tensor([[-1,0],[-1,1],[-1,2],[-1,3]], dtype=torch.int32)
    blocks = torch.tensor([[0]*3+[1]*9]*4, dtype=torch.long)
    batches = [(x[i:i+2], y[i:i+2], blocks[i:i+2]) for i in (0,2)]
    initial = {k:v.clone() for k,v in model.state_dict().items()}
    rng = torch.get_rng_state().clone()
    calls = []
    hook = model.register_forward_hook(lambda *args: calls.append(1))
    original = model.update_and_log_metrics.__func__
    with torch.inference_mode():
        expected, _, _ = model.process_predictions_and_compute_losses(model(x), y, blocks)
        valid = y.flatten().ne(-1)
        expected_predictions = expected[valid].argmax(-1).tolist()
        calls.clear()
        losses = [model.test_step(batch, i).clone() for i,batch in enumerate(batches)]
        ordinary_metric = model.test_metric.compute()['test_acc'].clone()
        assert len(calls) == 2
        model.test_metric.reset()
        calls.clear()
        with capture.capture_ordinary_classification(model,4) as rows:
            captured_losses = [model.test_step(batch,i).clone() for i,batch in enumerate(batches)]
            observed = rows.finalize()
        captured_metric = model.test_metric.compute()['test_acc'].clone()
    hook.remove()
    assert len(calls) == 2, 'Capture added an inference forward'
    assert observed['row_ids'] == list(range(4))
    assert observed['observed_labels'] == [0,1,2,3]
    assert observed['predictions'] == expected_predictions
    assert sum(p==y for p,y in zip(observed['predictions'],observed['observed_labels']))/4 == float(captured_metric)
    torch.testing.assert_close(ordinary_metric,captured_metric,rtol=0,atol=0)
    for a,b in zip(losses,captured_losses): torch.testing.assert_close(a,b,rtol=0,atol=0)
    assert model.update_and_log_metrics.__func__ is original
    assert '_ordinary_prediction_capture_active' not in vars(model)
    for key,value in model.state_dict().items(): torch.testing.assert_close(value,initial[key],rtol=0,atol=0)
    assert torch.equal(rng,torch.get_rng_state())
    assert not torch.cuda.is_initialized()
    records.append(dict(model=name,checkpoint=path.relative_to(ROOT).as_posix(),
        checkpoint_sha256=receipt['best_checkpoint_sha256'],model_class=type(model).__name__,
        training_objective=cfg.get('training_objective','classification'),
        dropout_consistency_alpha=float(cfg.get('dropout_consistency_alpha',0)),
        label_smoothing_epsilon=float(cfg.get('label_smoothing_epsilon',0)),
        synthetic_recordings=4,ordinary_forward_calls=2,captured_forward_calls=2,
        predictions_loss_metric_counts_state_rng_and_restore_match=True))
    print('[CPU_CAPTURE_TECHNICAL_PASS] '+name,flush=True)
result=dict(status='verified_CPU_synthetic_actual_registered_model_capture_integration',
    verified_at=datetime.now(timezone.utc).isoformat(),elapsed_seconds=time.perf_counter()-started,
    source_sha256=registered['source_sha256'],capture_sha256=registered['capture_sha256'],
    records=records,official_test_opened=False,real_dataset_constructed=False,
    Trainer_or_optimizer_used=False,CUDA_initialized=False,root_source_changed=False,
    final_evaluator_patch_applied=False,
    limitations='Technical synthetic inference only. No project accuracy result or final-cohort authorization; canonical training/config/source/data/test/cost integration remains required.')
TARGET.write_text(json.dumps(result,indent=2)+'\n',encoding='utf-8')
print('[CPU_CAPTURE_TECHNICAL_COMPLETE] '+str(len(records))+' models',flush=True)

def configuration(model,batch):
    with initialize_config_dir(config_dir=str(ROOT/'config'),version_base=None):
        cfg=compose(config_name='main',overrides=[f'experiment=SSC_{model}','+comparison=SSC',
            'random_seed=117',f'batch_size={batch}',f'dataset.batch_size={batch}',
            'dataset.num_workers=0','++dataset.spawn_safe_transforms=true',
            '++dataset.validation_only_data=true','++dataset.validation_batch_size=1024',
            'run_test=false','early_stopping=false','++calibration_only=true'])
    cfg=OmegaConf.create(OmegaConf.to_container(cfg,resolve=True))
    assert cfg.lr==.006 and cfg.n_epochs==40 and cfg.dropout==.15
    assert cfg.loss_agg=='mean_membrane_potentials' and cfg.dataset.num_classes==35
    assert cfg.l1.n_neurons==cfg.l2.n_neurons==720
    return cfg

def train_probe(model,cfg,cpu_samples):
    configured=model.configure_optimizers()
    optimizer=configured['optimizer'] if isinstance(configured,dict) else configured
    batch=tuple(t.cuda() for t in data.collate_fn(cpu_samples))
    assert batch[0].shape[:2]==(len(cpu_samples),300)
    times=[];losses=[]
    for step in range(5):
        optimizer.zero_grad(set_to_none=True);torch.cuda.synchronize()
        if step==2:torch.cuda.reset_peak_memory_stats()
        start=time.perf_counter()
        if cfg.get('training_objective')=='classification_dropout_consistency_v1':
            output,loss,diagnostics=model.training_terms(batch)
        else:
            output=model(batch[0]);_,loss,_=model.process_predictions_and_compute_losses(output,batch[1],batch[2])
        loss.backward();torch.nn.utils.clip_grad_norm_(model.parameters(),float(cfg.gradient_clip_val))
        optimizer.step();model.on_train_batch_end(None,batch,step)
        torch.cuda.synchronize();elapsed=time.perf_counter()-start
        losses.append(float(loss.detach()));assert math.isfinite(losses[-1])
        assert all(torch.isfinite(p.grad).all().item() for p in model.parameters() if p.grad is not None)
        assert all(torch.isfinite(p).all().item() for p in model.parameters())
        if step>=2:times.append(elapsed)
    reserved=torch.cuda.max_memory_reserved();allocated=torch.cuda.max_memory_allocated()
    return dict(warmup_steps=2,measured_steps=3,step_seconds=statistics.mean(times),
        step_time_samples=times,samples_per_second=len(cpu_samples)/statistics.mean(times),
        peak_allocated_gib=allocated/2**30,peak_reserved_gib=reserved/2**30,
        headroom_fraction=1-reserved/total,timesteps=300,updates_per_epoch=75466//len(cpu_samples),
        estimated_epoch_train_seconds=statistics.mean(times)*(75466//len(cpu_samples)),
        finite_losses_and_gradients=True,losses=losses,
        training_forward_passes=2 if cfg.get('training_objective')=='classification_dropout_consistency_v1' else 1)

def validation_pass(model,size):
    count=0;weighted=0.;predictions=[];torch.cuda.synchronize();start=time.perf_counter()
    with torch.inference_mode():
        for offset in range(0,len(cpu_val),size):
            x,y,blocks=(t.cuda() for t in data.collate_fn(cpu_val[offset:offset+size]))
            reduced,loss,_=model.process_predictions_and_compute_losses(model(x),y,blocks)
            mask=y.flatten().ne(-1);n=int(mask.sum());assert n==len(x)
            count+=n;weighted+=float(loss)*n;predictions.extend(reduced[mask].argmax(-1).cpu().tolist())
    torch.cuda.synchronize();elapsed=time.perf_counter()-start;assert count==9981
    return elapsed,weighted/count,predictions
